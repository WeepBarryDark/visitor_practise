import 'package:visitor_practise/core/models/image_config.dart';

class AdminDashboardController extends ChangeNotifier {

    }
  }
}

## 第三部分：完整的实际例子

### 完整的 Controller

```dart
import 'package:flutter/foundation.dart';
import 'package:flutter/services.dart';
import 'package:visitor_practise/core/models/image_config.dart';
import 'package:visitor_practise/core/mixins/ui_message_mixin.dart';
import 'package:visitor_practise/core/constants/server_link.dart';

class AdminDashboardController extends ChangeNotifier with UiMessageMixin {
  //------------------------------------------------
  // ✅ 图片配置
  //------------------------------------------------
  late ImageConfig logoConfig;
  late ImageConfig backgroundConfig;

  //------------------------------------------------
  // ✅ 其他属性
  //------------------------------------------------
  String? clientDisplayName;
  late final SiteItem currentSite;
  final TextEditingController adminPinCtrl = TextEditingController();
  String adminPin = '1234';
  bool showPreview = false;

  //------------------------------------------------
  // ✅ 初始化
  //------------------------------------------------
  Future<void> initialise({
    required Future<void> Function(String nextRoute) onAlreadyRedirect,
  }) async {
    final savedToken = await SecureStorageService.getAuthToken();
    if (savedToken == null || savedToken.isEmpty) {
      throw Exception('No token');
    }

    // 获取客户端信息
    final clientJson = await ApiService.fetchVisitorClient(savedToken);

    // ✅ 初始化图片配置
    logoConfig = ImageConfig(
      defaultUrl: ServerLink.defaultHeadLogo,
      customUrl: clientJson['logo'] as String?,
      bytes: await _loadLogoBytes(),
    );

    backgroundConfig = ImageConfig(
      defaultUrl: ServerLink.defaultBackgroup,
      customUrl: clientJson['background_image'] as String?,
    );

    clientDisplayName = clientJson['trading_name'];

    // 加载其他设置...
    await _loadVisitorRequirements();
    notifyListeners();
  }

  //------------------------------------------------
  // ✅ 加载 Logo bytes
  //------------------------------------------------
  Future<Uint8List?> _loadLogoBytes() async {
    try {
      final storedLogoBytes = await SecureStorageService.getClientLogoBytes();
      if (storedLogoBytes != null && storedLogoBytes.isNotEmpty) {
        return storedLogoBytes;
      }

      final ByteData data = await rootBundle.load(
        'lib/assets/images/WorxSafety_Logo_NoShadow.png'
      );
      return data.buffer.asUint8List();
    } catch (e) {
      debugPrint('Failed to load logo: $e');
      return null;
    }
  }

  //------------------------------------------------
  // ✅ 保存管理员密码（使用新的消息方式）
  //------------------------------------------------
  Future<void> onSaveAdminPin() async {
    final pin = adminPinCtrl.text.trim();

    if (pin.length < 4) {
      sendError('Password must be at least 4 characters');  // ✅ 简洁
      return;
    }

    try {
      await SecureStorageService.saveAdminPin(pin);
      sendSuccess('New admin pin saved');  // ✅ 简洁
    } catch (e) {
      sendError('Failed to save admin pin');  // ✅ 简洁
      debugPrint('Error: $e');
    }
  }

  //------------------------------------------------
  // ✅ 生成预览（使用新的消息方式）
  //------------------------------------------------
  Future<void> generatePreview() async {
    try {
      // 生成预览逻辑...
      showPreview = true;
      notifyListeners();

      sendInfo('Badge preview generated');  // ✅ 简洁
    } catch (e) {
      sendError('Failed to generate preview: $e');  // ✅ 简洁
    }
  }

  //------------------------------------------------
  // ✅ 确认并跳转到 Kiosk
  //------------------------------------------------
  Future<void> confirmToKiosk() async {
    try {
      await _saveVisitorRequirements();
      sendSuccess('Settings saved successfully');  // ✅ 简洁
    } catch (e) {
      sendError('Failed to save settings');  // ✅ 简洁
    }
  }
}
```

### 完整的 Page

```dart
import 'package:flutter/material.dart';
import 'package:visitor_practise/pages/admin_dashboard/controllers/admin_dashboard_controller.dart';
import 'package:visitor_practise/pages/admin_dashboard/widgets/admin_dashboard_main.dart';
import 'package:visitor_practise/shared_widgets/parent_widgets/background_image_parent.dart';

class AdminDashboardPage extends StatefulWidget {
  const AdminDashboardPage({super.key});

  @override
  State<AdminDashboardPage> createState() => _AdminDashboardPageState();
}

class _AdminDashboardPageState extends State<AdminDashboardPage> {
  late final AdminDashboardController _dashboardController;

  @override
  void initState() {
    super.initState();
    _dashboardController = AdminDashboardController();

    // ✅ 只需要这一行设置消息处理！
    _dashboardController.setupMessageHandler(context);

    // 初始化 controller
    _dashboardController.initialise(
      onAlreadyRedirect: (nextRoute) async => _handleAutoNavigation(nextRoute),
    );
  }

  @override
  Widget build(BuildContext context) {
    final width = MediaQuery.of(context).size.width;
    final maxBodyWidth = AppBreakpoints.getContentWidth(width);

    return AnimatedBuilder(
      animation: _dashboardController,
      builder: (context, _) {
        if (_dashboardController.isCheckingInitialDashboard) {
          return const LoadingCircleInterface();
        }

        return AppShell(
          title: 'Admin Dashboard',
          child: BackgroundImageParent(
            // ✅ 使用新的 ImageConfig
            webNotAsset: _dashboardController.backgroundConfig.isCustom,
            customBackgroundUrl: _dashboardController.backgroundConfig.currentUrl,
            mainWidget: AdminDashboardMain(
              adminDashboardController: _dashboardController,
              maxBodyWidth: maxBodyWidth,
            ),
          ),
        );
      },
    );
  }

  Future<void> _handleAutoNavigation(String nextRoute) async {
    try {
      Navigator.of(context).pushReplacementNamed(nextRoute);
    } catch (e) {
      if (!mounted) return;

      // ✅ 也可以直接在这里使用 extension
      context.showError(
        'Offline or Server Unreachable. Check Internet Connection.',
      );
    }
  }

  @override
  void dispose() {
    _dashboardController.disposeMessageHandler();  // 清理
    super.dispose();
  }
}
```
